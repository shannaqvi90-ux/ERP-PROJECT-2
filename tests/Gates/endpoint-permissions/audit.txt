# Reviewed endpoint-to-permission map of the audit module (G2). Every route of the module is
# listed with exactly the permission it declares. Each kind of record is filtered further by its own
# read permission in the record's company and branch (AuditReader.VisibleAsync): the history of a
# record tells everything it ever held, so the trail never shows more than the record does.
prefix /api/audit/

GET /api/audit/log audit.log.read # every recorded change the caller may read, a page at a time (the list query contract)
GET /api/audit/log/{id:guid} audit.log.read # one change of the log with its stored values; the same filter by kind of record, company and branch as the log
GET /api/audit/records/{id:guid} audit.log.read # one record's history (its form's history panel); the same filter by kind of record, company and branch as the log
