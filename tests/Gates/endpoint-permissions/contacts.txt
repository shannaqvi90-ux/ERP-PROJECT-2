# Reviewed endpoint-to-permission map of the contacts module (G2). Every endpoint whose route
# starts with a "prefix" below must be listed here with exactly the permission it declares, and
# every line must match an endpoint. A new endpoint, or a changed permission, fails the gate until
# this file is reviewed and changed with it: the reason after # says why that permission, and no
# broader one, guards the endpoint.
prefix /api/contacts/

GET /api/contacts/contacts contacts.contacts.read # the contacts list (the caller's workspace only)
GET /api/contacts/contacts/duplicates contacts.contacts.read # contacts that share a name, e-mail, number or TRN with one being typed: what the list itself shows to the same reader
GET /api/contacts/contacts/defaults contacts.contacts.read # what a new contact starts with: the country and the base currency of the caller's own working company, which a credit limit is stated in
GET /api/contacts/contacts/{id:guid} contacts.contacts.read # one contact: names, address, registration numbers, credit limit
POST /api/contacts/contacts contacts.contacts.create # a new organisation or person in the shared directory
PUT /api/contacts/contacts/{id:guid} contacts.contacts.update # changing a contact's details, its organisation or its credit limit; never its status
POST /api/contacts/contacts/{id:guid}/archive contacts.contacts.archive # taking a contact out of use is its own decision, apart from editing it
POST /api/contacts/contacts/{id:guid}/restore contacts.contacts.archive # bringing an archived contact back is the same decision reversed
